This file contains ONLY the schema additions/changes introduced in this release. The full
composed supergraph is generated by Rover CLI — do not hand-author it.

Cross-subgraph references: `payroll-subgraph` references the `Employee` entity below via
`@key(fields: "id")` (`@external` stub, since payroll only needs the id to attach
`Payslip`/`Notification` records). Implementation: `./servers/employee/src/`.

```graphql
extend schema @link(url: "https://specs.apollo.dev/federation/v2.14", import:["@key", "@shareable"])

# ─── Entity ───────────────────────────────────────────────────────────────────

"""
Employee entity. Owned by employee-subgraph.
"""
type Employee @key(fields: "id") {
  id: ID!
  employeeId: String!
  fullName: String!
  role: String!
  department: String!
  grossSalary: Float!
  supervisor: Employee
  createdAt: String!
}

# ─── Types ────────────────────────────────────────────────────────────────────

type RegisterEmployeeResult {
  employee: Employee!

  """Auto-generated temporary password for the employee's Authentik account. Shown once."""
  temporaryPassword: String!
}

input RegisterEmployeeInput {
  fullName: String!
  employeeId: String!
  role: String!
  department: String!
  grossSalary: Float!
  supervisorId: ID
}

input AssignSupervisorInput {
  employeeId: ID!
  supervisorId: ID!
}

# ─── Queries ──────────────────────────────────────────────────────────────────

extend type Query {
  employee(id: ID!): Employee
  employees(department: String, role: String): [Employee!]!
}

# ─── Mutations ────────────────────────────────────────────────────────────────

extend type Mutation {
  """
  Register a new employee: persists the record to Postgres and creates an Authentik
  account with an auto-generated temporary password (returned once). Rolls back the
  Postgres insert if Authentik account creation fails.
  """
  registerEmployee(input: RegisterEmployeeInput!): RegisterEmployeeResult!

  """
  Assign or reassign a supervisor for an existing employee. The supervisor must be an
  existing employee; an employee cannot be assigned as their own supervisor.
  """
  assignSupervisor(input: AssignSupervisorInput!): Employee!
}
```
