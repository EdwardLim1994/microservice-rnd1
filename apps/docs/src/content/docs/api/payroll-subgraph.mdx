This file contains ONLY the schema additions/changes introduced in this release. The full
composed supergraph is generated by Rover CLI — do not hand-author it.

Cross-subgraph references: extends the `Employee` entity owned by `employee-subgraph` via
`@key(fields: "id")`. Implementation: `./servers/payroll/src/`.

```graphql
extend schema @link(url: "https://specs.apollo.dev/federation/v2.14", import:["@key"])

# Owned by employee-subgraph. Federation v2 does NOT mark a stub entity's own @key field
# @external (that's Federation v1 syntax, reserved in v2 for fields pulled in via
# @requires/@provides) — marking it here breaks composition for any query that jumps to
# employee-subgraph through this key.
type Employee @key(fields: "id") {
  id: ID!
}

type Payslip @key(fields: "id") {
  id: ID!
  employee: Employee!
  month: Int!
  year: Int!
  minioObjectKey: String!
  generatedAt: String!
}

type GeneratePayslipsResult {
  generated: [Payslip!]!
  failed: [ID!]!
}

type Notification @key(fields: "id") {
  id: ID!
  employee: Employee!
  message: String!
  read: Boolean!
  createdAt: String!
}

input GeneratePayslipsInput {
  month: Int!
  year: Int!
}

extend type Query {
  notifications(employeeId: ID!): [Notification!]!
}

extend type Mutation {
  """
  Trigger monthly payslip generation for all employees (also fired automatically on the 1st of
  every month via CronDriver). Fetches employees from employee-subgraph, generates a PDF payslip
  per employee, uploads to Minio (FEAT-4 — no separate GraphQL surface, exercised only through
  this mutation), persists the Payslip record, and creates a Notification. Per-employee failures
  are collected in `failed` rather than aborting the whole run.
  """
  generatePayslips(input: GeneratePayslipsInput!): GeneratePayslipsResult!

  markNotificationRead(id: ID!): Notification!
}
```
